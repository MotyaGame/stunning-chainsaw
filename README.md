import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Создание данных о продажах товаров
data = {
    'Товар': ['Телефон', 'Ноутбук', 'Планшет', 'Смарт-часы', 'Наушники', 'Клавиатура', 'Мышь'],
    'Продажи': np.random.randint(50, 500, 7),  # Количество проданных единиц
    'Цена': np.random.randint(5000, 50000, 7)  # Цена товара в рублях
}

df = pd.DataFrame(data)

# Добавляем столбец с выручкой
df['Выручка'] = df['Продажи'] * df['Цена']

# Анализ данных
среднее_продаж = df['Продажи'].mean()
средневзвешенная_цена = np.average(df['Цена'], weights=df['Продажи'])
максимальная_выручка = df['Выручка'].max()
минимальная_выручка = df['Выручка'].min()

# Вывод результатов анализа
print("Среднее количество продаж:", среднее_продаж)
print("Средневзвешенная цена товара:", средневзвешенная_цена)
print("Максимальная выручка:", максимальная_выручка)
print("Минимальная выручка:", минимальная_выручка)

# Визуализация данных
plt.figure(figsize=(10, 5))
plt.bar(df['Товар'], df['Выручка'], color='skyblue')
plt.xlabel('Товар')
plt.ylabel('Выручка')
plt.title('Выручка по товарам')
plt.xticks(rotation=45)
plt.show()
